const getStateMonitoring = async (req, res) => {
  try {
    const [rows] = await pool.query(`
      SELECT
        s.state_id,
        s.state_name,

        sl.latitude,
        sl.longitude,

        dp.prediction_id,
        dp.prediction_time,
        dp.probability,
        dp.severity,
        dp.expected_start_time,
        dp.expected_end_time,
        dp.affected_area_km2,
        dp.affected_population,
        dp.confidence_score,

        dt.disaster_type_id,
        dt.type_name AS disaster_type,

        de.disaster_event_id,
        de.event_name,
        de.event_status

      FROM states s

      LEFT JOIN state_locations sl
        ON sl.state_id = s.state_id

      LEFT JOIN (
        SELECT dp1.*
        FROM disaster_predictions dp1
        INNER JOIN (
          SELECT
            disaster_event_id,
            MAX(prediction_id) AS latest_prediction_id
          FROM disaster_predictions
          GROUP BY disaster_event_id
        ) latest
          ON latest.latest_prediction_id = dp1.prediction_id
      ) dp
        ON dp.prediction_id = (
          SELECT MAX(dp2.prediction_id)
          FROM disaster_predictions dp2
          INNER JOIN disaster_events de2
            ON de2.disaster_event_id = dp2.disaster_event_id
          WHERE de2.state_id = s.state_id
        )

      LEFT JOIN disaster_events de
        ON de.disaster_event_id = dp.disaster_event_id

      LEFT JOIN disaster_types dt
        ON dt.disaster_type_id = de.disaster_type_id

      WHERE sl.state_location_id IS NOT NULL

      ORDER BY s.state_id
    `);

    return res.json({
      success: true,
      count: rows.length,
      states: rows
    });

  } catch (error) {
    console.error("Monitoring states error:", error);

    return res.status(500).json({
      success: false,
      message: error.message
    });
  }
};

const DISASTER_TYPE_MAP = {
    FLOOD: 1,
    CYCLONE: 2,
    EARTHQUAKE: 3,
    LANDSLIDE: 4,
    WILDFIRE: 5,
    DROUGHT: 6,
    TSUNAMI: 7

// FETCH 14 DAYS WEATHER
// ============================================================

async function fetchWeatherHistory(latitude, longitude) {

    const endDate = new Date();

    const startDate = new Date();
    startDate.setUTCDate(
        startDate.getUTCDate() - 13
    );

    const formatDate = (date) =>
        date.toISOString().slice(0, 10);

    const response = await axios.get(
        "https://archive-api.open-meteo.com/v1/archive",
        {
            params: {
                latitude,
                longitude,

                start_date: formatDate(startDate),
                end_date: formatDate(endDate),

                daily: [
                    "temperature_2m_mean",
                    "temperature_2m_max",
                    "temperature_2m_min",
                    "precipitation_sum",
                    "relative_humidity_2m_mean",
                    "surface_pressure_mean",
                    "wind_speed_10m_mean",
                    "wind_speed_10m_max",
                    "wind_direction_10m_dominant"
                ].join(","),

                timezone: "UTC"
            },

            timeout: 60000
        }
    );

    return response.data;
}


// ============================================================
// BUILD MODEL FEATURES
// ============================================================

function buildFeatures(weather) {

    const daily = weather.daily;

    if (!daily || !daily.time || daily.time.length === 0) {
        throw new Error(
            "No historical weather data returned"
        );
    }

    const last = daily.time.length - 1;

    const get = (array, index = last) =>
        Number(array?.[index] ?? 0);

    const precipitation =
        daily.precipitation_sum || [];

    const humidity =
        daily.relative_humidity_2m_mean || [];

    const temperature =
        daily.temperature_2m_mean || [];

    const wind =
        daily.wind_speed_10m_mean || [];

    const windMax =
        daily.wind_speed_10m_max || [];

    const pressure =
        daily.surface_pressure_mean || [];


    const sumWindow = (array, window) => {

        const start =
            Math.max(0, array.length - window);

        return array
            .slice(start)
            .reduce(
                (sum, value) =>
                    sum + Number(value || 0),
                0
            );
    };


    const meanWindow = (array, window) => {

        const start =
            Math.max(0, array.length - window);

        const values =
            array
                .slice(start)
                .map(Number)
                .filter(Number.isFinite);

        if (!values.length) {
            return 0;
        }

        return (
            values.reduce(
                (sum, value) => sum + value,
                0
            ) / values.length
        );
    };


    const maxWindow = (array, window) => {

        const start =
            Math.max(0, array.length - window);

        const values =
            array
                .slice(start)
                .map(Number)
                .filter(Number.isFinite);

        return values.length
            ? Math.max(...values)
            : 0;
    };


    const currentTemperature =
        get(daily.temperature_2m_mean);

    const currentTemperatureMax =
        get(daily.temperature_2m_max);

    const currentTemperatureMin =
        get(daily.temperature_2m_min);


    const currentPrecipitation =
        get(precipitation);


    const currentHumidity =
        get(humidity);


    const currentPressure =
        get(pressure);


    const previousPressure =
        pressure.length >= 2
            ? Number(pressure[pressure.length - 2] || 0)
            : currentPressure;


    const pressure3DaysAgo =
        pressure.length >= 4
            ? Number(pressure[pressure.length - 4] || 0)
            : currentPressure;


    return {

        temperature_2m_mean:
            currentTemperature,

        temperature_2m_max:
            currentTemperatureMax,

        temperature_2m_min:
            currentTemperatureMin,

        temperature_range:
            currentTemperatureMax -
            currentTemperatureMin,


        precipitation_sum:
            currentPrecipitation,


        rainfall_1d:
            currentPrecipitation,

        rainfall_3d:
            sumWindow(
                precipitation,
                3
            ),

        rainfall_7d:
            sumWindow(
                precipitation,
                7
            ),

        rainfall_14d:
            sumWindow(
                precipitation,
                14
            ),


        rainfall_3d_mean:
            meanWindow(
                precipitation,
                3
            ),

        rainfall_7d_mean:
            meanWindow(
                precipitation,
                7
            ),


        relative_humidity_2m_mean:
            currentHumidity,

        humidity_3d_mean:
            meanWindow(
                humidity,
                3
            ),

        humidity_7d_mean:
            meanWindow(
                humidity,
                7
            ),


        surface_pressure_mean:
            currentPressure,

        pressure_change_1d:
            currentPressure -
            previousPressure,

        pressure_change_3d:
            currentPressure -
            pressure3DaysAgo,


        wind_speed_10m_mean:
            get(
                daily.wind_speed_10m_mean
            ),

        wind_speed_10m_max:
            get(
                daily.wind_speed_10m_max
            ),

        wind_3d_mean:
            meanWindow(
                wind,
                3
            ),

        wind_7d_mean:
            meanWindow(
                wind,
                7
            ),

        wind_max_7d:
            maxWindow(
                windMax,
                7
            ),

        wind_direction_10m_dominant:
            get(
                daily.wind_direction_10m_dominant
            )
    };
}


// ============================================================
// ESTIMATE AFFECTED POPULATION
// ============================================================

function estimateAffectedPopulation(
    probability,
    severity
) {

    const basePopulation = {
        LOW: 5000,
        MEDIUM: 10000,
        HIGH: 25000,
        CRITICAL: 50000
    };

    const base =
        basePopulation[severity] || 5000;

    return Math.round(
        base * probability
    );
}

// ============================================================
// CREATE / UPDATE PREDICTION
// ============================================================

async function runLivePrediction({
    latitude,
    longitude,
    stateId,
    districtId = null
}) {

    latitude = Number(latitude);
    longitude = Number(longitude);
    stateId = Number(stateId);

    if (
        !Number.isFinite(latitude) ||
        !Number.isFinite(longitude)
    ) {
        throw new Error(
            "Valid latitude and longitude are required"
        );
    }

    if (!Number.isInteger(stateId)) {
        throw new Error(
            "Valid stateId is required"
        );
    }


    // --------------------------------------------------------
    // 1. State
    // --------------------------------------------------------

    const state =
        await getState(stateId);


    // --------------------------------------------------------
    // 2. Weather
    // --------------------------------------------------------

    const weather =
        await fetchWeatherHistory(
            latitude,
            longitude
        );


    // --------------------------------------------------------
    // 3. Features
    // --------------------------------------------------------

    const features =
        buildFeatures(weather);


    // --------------------------------------------------------
    // 4. ML prediction
    // --------------------------------------------------------

    const mlResponse =
        await axios.post(
            `${ML_SERVICE_URL}/predict`,
            features,
            {
                timeout: 30000
            }
        );


    const prediction =
        mlResponse.data;


    if (!prediction.success) {
        throw new Error(
            "ML prediction failed"
        );
    }


    // --------------------------------------------------------
    // 5. Disaster type
    // --------------------------------------------------------

    const predictedDisaster =
        String(
            prediction.predicted_disaster
        ).toUpperCase();


    // ============================================================
// NORMAL = NO DISASTER DETECTED
// ============================================================

if (predictedDisaster === "NORMAL") {

    return {

        success: true,

        disasterDetected: false,

        message:
            "No disaster risk detected for the selected location.",

        state: {
            stateId:
                state.state_id,

            stateName:
                state.state_name
        },

        location: {
            latitude,
            longitude
        },

        prediction: {
            disaster: "NORMAL",

            probability:
                prediction.probability,

            severity: "LOW",

            confidence:
                prediction.probability,

            classProbabilities:
                prediction.class_probabilities
        },

        affectedPopulation: 0,

        resourceRequirements: {
            success: true,
            requirements: [],
            message:
                "No resources required because no disaster was detected."
        },

        weatherFeatures:
            features
    };
}


// ============================================================
// ACTUAL DISASTER
// ============================================================

const disasterTypeId =
    DISASTER_TYPE_MAP[
        predictedDisaster
    ];


if (!disasterTypeId) {

    throw new Error(
        `Unsupported disaster type: ${predictedDisaster}`
    );

}


    // --------------------------------------------------------
    // 6. Affected population
    // --------------------------------------------------------

    const affectedPopulation =
        estimateAffectedPopulation(
            prediction.probability,
            prediction.severity
        );


    // --------------------------------------------------------
    // 7. Database transaction
    // --------------------------------------------------------

    const connection =
        await pool.getConnection();


    try {

        await connection.beginTransaction();


        // ----------------------------------------------------
        // Disaster event
        // ----------------------------------------------------

        const eventName =
            `${predictedDisaster} Prediction - ${state.state_name}`;


        const [eventResult] =
            await connection.query(
                `
                INSERT INTO disaster_events
                (
                    disaster_type_id,
                    state_id,
                    district_id,
                    event_name,
                    event_status
                )
                VALUES (?, ?, ?, ?, 'PREDICTED')
                `,
                [
                    disasterTypeId,
                    stateId,
                    districtId,
                    eventName
                ]
            );


        const disasterEventId =
            eventResult.insertId;


        // ----------------------------------------------------
        // Prediction
        // ----------------------------------------------------

        const predictionTime =
            new Date()
                .toISOString()
                .slice(0, 19)
                .replace("T", " ");


        const [predictionResult] =
            await connection.query(
                `
                INSERT INTO disaster_predictions
                (
                    disaster_event_id,
                    prediction_model_id,
                    prediction_time,
                    probability,
                    severity,
                    expected_start_time,
                    expected_end_time,
                    affected_area_km2,
                    affected_population,
                    confidence_score,
                    prediction_details
                )
                VALUES
                (
                    ?,
                    1,
                    ?,
                    ?,
                    ?,
                    NULL,
                    NULL,
                    NULL,
                    ?,
                    ?,
                    ?
                )
                `,
                [
                    disasterEventId,
                    predictionTime,
                    prediction.probability * 100,
                    prediction.severity,
                    affectedPopulation,
                    prediction.probability * 100,
                    JSON.stringify({
                        source: "LIVE_WEATHER_ML",
                        latitude,
                        longitude,
                        stateId,
                        districtId,
                        predictedDisaster,
                        classProbabilities:
                            prediction.class_probabilities,
                        weatherFeatures:
                            features
                    })
                ]
            );


        const predictionId =
            predictionResult.insertId;


        // ----------------------------------------------------
        // Prediction area
        // ----------------------------------------------------

        if (districtId) {

            await connection.query(
                `
                INSERT INTO prediction_areas
                (
                    prediction_id,
                    district_id,
                    risk_level,
                    population_at_risk
                )
                VALUES (?, ?, ?, ?)
                `,
                [
                    predictionId,
                    districtId,
                    prediction.severity,
                    affectedPopulation
                ]
            );

        }


        await connection.commit();


        // ----------------------------------------------------
        // Resource requirements
        // ----------------------------------------------------

        const {
            generateResourceRequirements
        } = require(
            "./resourceRequirementService"
        );


        let resourceRequirements = null;


        try {

            resourceRequirements =
                await generateResourceRequirements(
                    predictionId
                );

        } catch (resourceError) {

            console.error(
                "Resource requirement generation failed:",
                resourceError.message
            );

            resourceRequirements = {
                success: false,
                message:
                    resourceError.message
            };
        }


        return {

            success: true,

            predictionId,

            disasterEventId,

            state: {
                stateId:
                    state.state_id,

                stateName:
                    state.state_name
            },

            location: {
                latitude,
                longitude
            },

            prediction: {
                disaster:
                    predictedDisaster,

                probability:
                    prediction.probability,

                severity:
                    prediction.severity,

                confidence:
                    prediction.probability,

                classProbabilities:
                    prediction.class_probabilities
            },

            affectedPopulation,

            resourceRequirements,

            weatherFeatures:
                features
        };


    } catch (error) {

        await connection.rollback();

        throw error;

    } finally {

        connection.release();

    }
}


module.exports = {
    runLivePrediction
};

/*
|--------------------------------------------------------------------------
| CALCULATE RESOURCE REQUIREMENTS
|--------------------------------------------------------------------------
|
| This function:
|
| 1. Gets prediction details
| 2. Finds matching resource requirement rules
| 3. Calculates required quantity
| 4. Checks warehouse inventory
| 5. Calculates available quantity
| 6. Calculates shortage
|
| IMPORTANT:
| This function DOES NOT modify the database.
|
*/

async function calculateResourceRequirements(predictionId) {

  // ---------------------------------------------------------
  // 1. Get prediction details
  // ---------------------------------------------------------

  const [predictionRows] = await pool.query(
    `
    SELECT
        dp.prediction_id,
        dp.probability,
        dp.severity,
        dp.affected_population,

        de.disaster_event_id,
        de.disaster_type_id,

        dt.disaster_name,

        de.state_id,
        s.state_name

    FROM disaster_predictions dp

    JOIN disaster_events de
        ON de.disaster_event_id = dp.disaster_event_id

    JOIN disaster_types dt
        ON dt.disaster_type_id = de.disaster_type_id

    JOIN states s
        ON s.state_id = de.state_id

    WHERE dp.prediction_id = ?
    `,
    [predictionId]
  );


  if (predictionRows.length === 0) {

    throw new Error(
      "Prediction not found"
    );

  }


  const prediction = predictionRows[0];


  // ---------------------------------------------------------
  // 2. Validate affected population
  // ---------------------------------------------------------

  const affectedPopulation = Number(
    prediction.affected_population || 0
  );


  if (affectedPopulation <= 0) {

    throw new Error(
      "Prediction does not contain a valid affected population"
    );

  }


  // ---------------------------------------------------------
  // 3. Find matching resource requirement rules
  // ---------------------------------------------------------

  const [rules] = await pool.query(
    `
    SELECT
        rrr.disaster_type_id,
        rrr.resource_id,
        rrr.severity,
        rrr.quantity_per_person,
        rrr.fixed_quantity,

        r.resource_name,
        r.unit

    FROM resource_requirement_rules rrr

    JOIN resources r
        ON r.resource_id = rrr.resource_id

    WHERE rrr.disaster_type_id = ?
      AND rrr.severity = ?

    ORDER BY r.resource_id
    `,
    [
      prediction.disaster_type_id,
      prediction.severity
    ]
  );


  // ---------------------------------------------------------
  // No matching rules
  // ---------------------------------------------------------

  if (rules.length === 0) {

    return {
      prediction,
      requirements: [],
      message:
        "No resource requirement rules configured for this disaster type and severity"
    };

  }


  // ---------------------------------------------------------
  // 4. Calculate requirements
  // ---------------------------------------------------------

  const requirements = [];


  for (const rule of rules) {

    let quantityRequired = 0;


    // -------------------------------------------------------
    // Quantity based on affected population
    // -------------------------------------------------------

    if (rule.quantity_per_person !== null) {

      quantityRequired =
        affectedPopulation *
        Number(rule.quantity_per_person);

    }


    // -------------------------------------------------------
    // Fixed quantity
    // -------------------------------------------------------

    else if (rule.fixed_quantity !== null) {

      quantityRequired =
        Number(rule.fixed_quantity);

    }


    quantityRequired = Number(
      quantityRequired.toFixed(3)
    );


    // -------------------------------------------------------
    // 5. Get inventory from all warehouses
    // -------------------------------------------------------

    const [inventoryRows] = await pool.query(
      `
      SELECT
          w.warehouse_id,
          w.warehouse_name,

          w.state_id,
          s.state_name,

          wr.quantity_available,
          wr.quantity_reserved,

          (
              wr.quantity_available -
              wr.quantity_reserved
          ) AS usable_quantity

      FROM warehouse_resources wr

      JOIN warehouses w
          ON w.warehouse_id = wr.warehouse_id

      JOIN states s
          ON s.state_id = w.state_id

      WHERE wr.resource_id = ?

      ORDER BY usable_quantity DESC
      `,
      [rule.resource_id]
    );


    // -------------------------------------------------------
    // 6. Calculate total usable inventory
    // -------------------------------------------------------

    const totalAvailable =
      inventoryRows.reduce(
        (total, row) =>
          total +
          Number(row.usable_quantity || 0),
        0
      );


    // -------------------------------------------------------
    // 7. Calculate shortage
    // -------------------------------------------------------

    const shortage = Math.max(
      quantityRequired -
      totalAvailable,
      0
    );


    // -------------------------------------------------------
    // 8. Add requirement to result
    // -------------------------------------------------------

    requirements.push({

      resource_id:
        rule.resource_id,

      resource_name:
        rule.resource_name,

      unit:
        rule.unit,

      quantity_required:
        quantityRequired,

      quantity_available:
        Number(
          totalAvailable.toFixed(3)
        ),

      quantity_shortage:
        Number(
          shortage.toFixed(3)
        ),

      inventory:
        inventoryRows.map((row) => ({

          warehouse_id:
            row.warehouse_id,

          warehouse_name:
            row.warehouse_name,

          state_id:
            row.state_id,

          state_name:
            row.state_name,

          quantity_available:
            Number(
              row.quantity_available
            ),

          quantity_reserved:
            Number(
              row.quantity_reserved
            ),

          usable_quantity:
            Number(
              row.usable_quantity
            )

        }))

    });

  }


  // ---------------------------------------------------------
  // 9. Return calculated requirements
  // ---------------------------------------------------------

  return {

    prediction,

    requirements

  };

}


/*
|--------------------------------------------------------------------------
| GENERATE / SAVE RESOURCE REQUIREMENTS
|--------------------------------------------------------------------------
|
| This function:
|
| 1. Calculates the resource requirements
| 2. Starts a database transaction
| 3. Removes previous generated requirements
| 4. Saves the latest calculated requirements
| 5. Commits the transaction
|
| This allows predictions such as Prediction #2,
| which currently has no rows in predicted_resource_requirements,
| to get their calculated requirements stored in the database.
|
*/

async function generateResourceRequirements(predictionId) {

  // ---------------------------------------------------------
  // 1. Calculate requirements first
  // ---------------------------------------------------------

  const result =
    await calculateResourceRequirements(
      predictionId
    );


  // ---------------------------------------------------------
  // 2. Make sure requirements exist
  // ---------------------------------------------------------

  if (
    !result.requirements ||
    result.requirements.length === 0
  ) {

    return {

      success: false,

      prediction:
        result.prediction,

      requirements: [],

      message:
        result.message ||
        "No resource requirements could be generated"

    };

  }


  // ---------------------------------------------------------
  // 3. Get database connection
  // ---------------------------------------------------------

  const connection =
    await pool.getConnection();


  try {

    // -------------------------------------------------------
    // 4. Start transaction
    // -------------------------------------------------------

    await connection.beginTransaction();


    // -------------------------------------------------------
    // 5. Remove previously generated requirements
    // -------------------------------------------------------
    //
    // This prevents duplicate rows if the user generates
    // requirements for the same prediction more than once.
    //
    // Existing resource requests are NOT deleted.
    //
    // -------------------------------------------------------

    await connection.query(
      `
      DELETE FROM predicted_resource_requirements
      WHERE prediction_id = ?
      `,
      [predictionId]
    );


    // -------------------------------------------------------
    // 6. Insert calculated requirements
    // -------------------------------------------------------

    const savedRequirements = [];


    for (
      const requirement
      of result.requirements
    ) {

      const [insertResult] =
        await connection.query(
          `
          INSERT INTO predicted_resource_requirements
          (
              prediction_id,
              resource_id,
              quantity_required,
              quantity_available,
              quantity_shortage
          )

          VALUES
          (?, ?, ?, ?, ?)
          `,
          [

            predictionId,

            requirement.resource_id,

            requirement.quantity_required,

            requirement.quantity_available,

            requirement.quantity_shortage

          ]
        );


      savedRequirements.push({

        predicted_requirement_id:
          insertResult.insertId,

        prediction_id:
          predictionId,

        resource_id:
          requirement.resource_id,

        resource_name:
          requirement.resource_name,

        unit:
          requirement.unit,

        quantity_required:
          requirement.quantity_required,

        quantity_available:
          requirement.quantity_available,

        quantity_shortage:
          requirement.quantity_shortage,

        inventory:
          requirement.inventory

      });

    }


    // -------------------------------------------------------
    // 7. Commit transaction
    // -------------------------------------------------------

    await connection.commit();


    // -------------------------------------------------------
    // 8. Return result
    // -------------------------------------------------------

    return {

      success: true,

      prediction:
        result.prediction,

      requirements:
        savedRequirements,

      message:
        "Resource requirements generated and saved successfully"

    };


  } catch (error) {

    // -------------------------------------------------------
    // Rollback if anything fails
    // -------------------------------------------------------

    await connection.rollback();

    throw error;


  } finally {

    // -------------------------------------------------------
    // Release connection
    // -------------------------------------------------------

    connection.release();

  }

}

            "predicted_disaster": str(
                predicted_disaster
            ),
            "probability": round(
                probability,
                6
            ),
            "severity": severity,
            "class_probabilities":
                class_probabilities
        }

    except Exception as e:

        raise HTTPException(
            status_code=500,
            detail=str(e)

@app.post("/optimize")

    candidates = [
        candidate.model_dump()
        for candidate in request.candidates
    ]
