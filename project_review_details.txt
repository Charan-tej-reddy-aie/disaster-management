import {
  MapContainer,
  TileLayer,
  CircleMarker,
  Marker,
  Popup,
} from "react-leaflet";

import L from "leaflet";
import "leaflet/dist/leaflet.css";

delete L.Icon.Default.prototype._getIconUrl;

L.Icon.Default.mergeOptions({
  iconRetinaUrl:
    "https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.9.4/images/marker-icon-2x.png",

  iconUrl:
    "https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.9.4/images/marker-icon.png",

  shadowUrl:
    "https://cdnjs.cloudflare.com/ajax/libs/leaflet/1.9.4/images/marker-shadow.png",
});

function getSeverityColor(severity) {
  switch (severity) {
    case "CRITICAL":
      return "#dc2626";

    case "HIGH":
      return "#f97316";

    case "MEDIUM":
      return "#eab308";

    case "LOW":
      return "#22c55e";

    default:
      return "#64748b";
  }
}

export default function DisasterMap({
  states = [],
  predictions = [],
  warehouses = [],
}) {
  const defaultCenter = [12.9716, 78.6569];

  return (
    <div className="h-[520px] w-full overflow-hidden rounded-b-xl">

      <MapContainer
        center={defaultCenter}
        zoom={6}
        scrollWheelZoom={true}
        className="h-full w-full"
      >

        <TileLayer
          attribution="&copy; OpenStreetMap contributors"
          url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
        />

        {/* States */}
        {states.map((state) => (
          <CircleMarker
            key={`state-${state.state_id}`}
            center={[
              Number(state.latitude),
              Number(state.longitude),
            ]}
            radius={8}
            pathOptions={{
              color: "#2563eb",
              fillColor: "#3b82f6",
              fillOpacity: 0.7,
            }}
          >
            <Popup>
              <div className="min-w-[150px]">
                <h3 className="font-bold">
                  {state.state_name}
                </h3>

                <p className="mt-1 text-xs text-slate-500">
                  State monitoring location
                </p>
              </div>
            </Popup>
          </CircleMarker>
        ))}

        {/* Predictions */}
        {predictions.map((prediction) => {
          if (
            !prediction.state_latitude ||
            !prediction.state_longitude
          ) {
            return null;
          }

          const color = getSeverityColor(
            prediction.severity
          );

          return (
            <CircleMarker
              key={`prediction-${prediction.prediction_id}`}
              center={[
                Number(prediction.state_latitude),
                Number(prediction.state_longitude),
              ]}
              radius={
                prediction.severity === "CRITICAL"
                  ? 24
                  : prediction.severity === "HIGH"
                    ? 19
                    : 14
              }
              pathOptions={{
                color,
                fillColor: color,
                fillOpacity: 0.35,
                weight: 3,
              }}
            >
              <Popup>
                <div className="min-w-[210px]">

                  <h3 className="text-base font-bold">
                    {prediction.disaster_name}
                  </h3>

                  <p className="mt-1 text-sm">
                    {prediction.state_name}
                  </p>

                  <div className="mt-3 space-y-1 text-xs">

                    <p>
                      <strong>Severity:</strong>{" "}
                      {prediction.severity}
                    </p>

                    <p>
                      <strong>Probability:</strong>{" "}
                      {(
                        Number(prediction.probability) * 100
                      ).toFixed(0)}
                      %
                    </p>

                    <p>
                      <strong>Affected Area:</strong>{" "}
                      {Number(
                        prediction.affected_area_km2 || 0
                      ).toFixed(0)}
                      {" "}km²
                    </p>

                    <p>
                      <strong>Population:</strong>{" "}
                      {Number(
                        prediction.affected_population || 0
                      ).toLocaleString()}
                    </p>

                  </div>

                </div>
              </Popup>
            </CircleMarker>
          );
        })}

        {/* Warehouses */}
        {warehouses.map((warehouse) => (
          <Marker
            key={`warehouse-${warehouse.warehouse_id}`}
            position={[
              Number(warehouse.latitude),
              Number(warehouse.longitude),
            ]}
          >
            <Popup>
              <div className="min-w-[220px]">

                <h3 className="font-bold">
                  {warehouse.warehouse_name}
                </h3>

                <p className="mt-1 text-xs text-slate-500">
                  {warehouse.state_name}
                </p>

                <div className="mt-3">
                  <p className="text-xs text-slate-600">
                    Usable inventory
                  </p>

                  <p className="text-lg font-bold text-green-600">
                    {Number(
                      warehouse.total_usable_inventory || 0
                    ).toLocaleString()}
                  </p>
                </div>

              </div>
            </Popup>
          </Marker>
        ))}

      </MapContainer>

    </div>
  );
}

function getSeverityClass(severity) {
  switch (severity) {
    case "CRITICAL":

    case "HIGH":
      return "bg-orange-100 text-orange-700 border-orange-200";

    case "LOW":

/* =========================================================
   DATE FORMATTER
========================================================= */

function formatDate(date) {
  if (!date) return "Not available";

  return new Date(date).toLocaleString("en-IN", {
    dateStyle: "medium",
    timeStyle: "short",
  });
}

/* =========================================================

========================================================= */

function PredictionCard({ prediction, onClick }) {
  const probability = Number(prediction.probability || 0);

  /*
    Probability is stored as:
    0.93 → 93%

    Confidence can be stored as:
    0.91 → 91%
    OR
    91 → 91%

    This handles both cases.
  */
  const rawConfidence = Number(
    prediction.confidence_score || 0
  );

  const confidence =
    rawConfidence <= 1
      ? rawConfidence * 100
      : rawConfidence;

  return (
    <div
      onClick={() => onClick(prediction)}
      className="cursor-pointer rounded-2xl border border-gray-200 bg-white p-5 shadow-sm transition hover:-translate-y-1 hover:shadow-lg"
    >
      {/* Header */}
      <div className="flex items-start justify-between">

            <AlertTriangle
              size={20}
              className="text-orange-500"

            <h3 className="text-lg font-bold text-gray-900">
              {prediction.disaster_name}

          </div>

          <div className="mt-2 flex items-center gap-2 text-sm text-gray-500">
            <MapPin size={15} />

            {prediction.state_name || "Unknown State"}

                    Shortage

                    </div>

                          <div
                            key={
                              warehouse.warehouse_id

                            className="flex flex-col gap-2 rounded-lg bg-gray-50 p-3 sm:flex-row sm:items-center sm:justify-between"

                                {

                                }
                              </p>
                            </div>

                            <div className="text-left sm:text-right">
                              <p className="text-sm font-semibold text-gray-800">
                                Usable:{" "}
                                {Number(
                                  warehouse.usable_quantity ||
                                    0
                                ).toLocaleString()}{" "}
                                {resource.unit}
                              </p>

                              <p className="text-xs text-gray-500">
                                Available:{" "}
                                {Number(
                                  warehouse.quantity_available ||
                                    0
                                ).toLocaleString()}{" "}
                                | Reserved:{" "}
                                {Number(
                                  warehouse.quantity_reserved ||
                                    0

                          </div>

                    </div>

            </div>

      </div>

      {requirements.some(
        (resource) =>

      ) && (
        <div className="mt-5 rounded-xl border border-orange-200 bg-orange-50 p-4">
          <div className="flex items-start gap-3">
            <AlertTriangle

                allocation.
              </p>
            </div>
          </div>
        </div>

    </div>

function PredictionModal({
  prediction,
  resourceRequirements,
  requirementsLoading,
  requirementsError,
  onClose,
  onCreateRequest,
}) {
  if (!prediction) return null;

  const probability = Number(
    prediction.probability || 0
  );

  const rawConfidence = Number(
    prediction.confidence_score || 0
  );

    rawConfidence <= 1
      ? rawConfidence * 100
      : rawConfidence;

  const hasShortage =
    resourceRequirements &&
    resourceRequirements.some(
      (resource) =>
        Number(resource.quantity_shortage || 0) > 0
    );

    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4">

import axios from "axios";

const api = axios.create({
  baseURL: "http://localhost:5000/api",
  headers: {
    "Content-Type": "application/json",
  },
});

api.interceptors.request.use(
  (config) => {
    const token = localStorage.getItem("token");

    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }

    return config;
  },
  (error) => Promise.reject(error)
);

api.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response?.status === 401) {
      localStorage.removeItem("token");
      localStorage.removeItem("user");

      window.location.href = "/login";
    }

    return Promise.reject(error);
  }

const pool = require("../config/database");

async function getDashboardData() {

  // --------------------------------------------------
  // 1. Prediction statistics
  // --------------------------------------------------

  const [[predictionStats]] = await pool.query(`
    SELECT
      COUNT(*) AS total_predictions,

      SUM(
        CASE
          WHEN severity = 'CRITICAL'
          THEN 1
          ELSE 0
        END
      ) AS critical_predictions,

      SUM(
        CASE
          WHEN severity = 'HIGH'
          THEN 1
          ELSE 0
        END
      ) AS high_predictions,

      SUM(
        CASE
          WHEN severity = 'MEDIUM'
          THEN 1
          ELSE 0
        END
      ) AS medium_predictions,

      SUM(
        CASE
          WHEN severity = 'LOW'
          THEN 1
          ELSE 0
        END
      ) AS low_predictions

    FROM disaster_predictions
  `);


  // --------------------------------------------------
  // 2. Warehouse statistics
  // --------------------------------------------------

  const [[warehouseStats]] = await pool.query(`
    SELECT
      COUNT(*) AS total_warehouses
    FROM warehouses
  `);


  // --------------------------------------------------
  // 3. Pending resource requests
  // --------------------------------------------------

  const [[requestStats]] = await pool.query(`
    SELECT
      COUNT(*) AS pending_requests
    FROM resource_requests
    WHERE request_status IN (
      'PENDING_STATE_APPROVAL',
      'PENDING_NATIONAL_APPROVAL'
    )
  `);


  // --------------------------------------------------
  // 4. Disaster predictions
  // --------------------------------------------------

  const [predictions] = await pool.query(`
    SELECT
      dp.prediction_id,
      dp.disaster_event_id,
      dp.prediction_time,
      dp.probability,
      dp.severity,
      dp.expected_start_time,
      dp.expected_end_time,
      dp.affected_area_km2,
      dp.affected_population,
      dp.confidence_score,

      dt.disaster_type_id,
      dt.disaster_name,

      de.event_name,
      de.event_status,

      s.state_id,
      s.state_name,

      sl.latitude AS state_latitude,
      sl.longitude AS state_longitude

    FROM disaster_predictions dp

    JOIN disaster_events de
      ON de.disaster_event_id = dp.disaster_event_id

    JOIN disaster_types dt
      ON dt.disaster_type_id = de.disaster_type_id

    JOIN states s
      ON s.state_id = de.state_id

    LEFT JOIN state_locations sl
      ON sl.state_id = s.state_id

    ORDER BY dp.prediction_time DESC

    LIMIT 50
  `);


  // --------------------------------------------------
  // 5. Warehouses
  // --------------------------------------------------

  const [warehouses] = await pool.query(`
    SELECT
      w.warehouse_id,
      w.warehouse_name,
      w.state_id,
      s.state_name,
      w.district_id,
      w.latitude,
      w.longitude,
      w.address,

      COALESCE(
        SUM(wr.quantity_available - wr.quantity_reserved),
        0
      ) AS total_usable_inventory

    FROM warehouses w

    JOIN states s
      ON s.state_id = w.state_id

    LEFT JOIN warehouse_resources wr
      ON wr.warehouse_id = w.warehouse_id

    GROUP BY
      w.warehouse_id,
      w.warehouse_name,
      w.state_id,
      s.state_name,
      w.district_id,
      w.latitude,
      w.longitude,
      w.address

    ORDER BY w.warehouse_id
  `);


  // --------------------------------------------------
  // 6. State locations
  // --------------------------------------------------

  const [states] = await pool.query(`
    SELECT
      sl.state_location_id,
      sl.state_id,
      s.state_name,
      sl.latitude,
      sl.longitude

    FROM state_locations sl

    JOIN states s
      ON s.state_id = sl.state_id

    ORDER BY s.state_name
  `);


  return {
    statistics: {
      predictions: Number(
        predictionStats.total_predictions || 0
      ),

      critical: Number(
        predictionStats.critical_predictions || 0
      ),

      high: Number(
        predictionStats.high_predictions || 0
      ),

      medium: Number(
        predictionStats.medium_predictions || 0
      ),

      low: Number(
        predictionStats.low_predictions || 0
      ),

      warehouses: Number(
        warehouseStats.total_warehouses || 0
      ),

      pendingRequests: Number(
        requestStats.pending_requests || 0
      ),
    },

    predictions,

    warehouses,

    states,
  };
}

const jwt = require("jsonwebtoken");

const authenticate = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader) {
      return res.status(401).json({
        success: false,
        message: "Authorization token is required",
      });
    }

    const parts = authHeader.split(" ");

    if (parts.length !== 2 || parts[0] !== "Bearer") {
      return res.status(401).json({
        success: false,
        message: "Invalid authorization format",
      });
    }

    const token = parts[1];

    const decoded = jwt.verify(
      token,
      process.env.JWT_SECRET
    );

    req.user = decoded;

    next();

  } catch (error) {
    console.error("Authentication error:", error.message);

    return res.status(401).json({
      success: false,
      message: "Invalid or expired token",
    });
  }
};

module.exports = {
  authenticate,
};

    (wr.quantity_available - wr.quantity_reserved) AS usable_quantity
FROM warehouse_resources wr
JOIN warehouses w
    ON w.warehouse_id = wr.warehouse_id
JOIN resources r
    ON r.resource_id = wr.resource_id
WHERE (wr.quantity_available - wr.quantity_reserved) > 0
ORDER BY w.warehouse_id, r.resource_id;
